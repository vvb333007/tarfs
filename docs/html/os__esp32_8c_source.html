122/*
123 * TARFS uses a single recursive mutex, if one is available on the platform.
124 * If recursive mutexes are not available, this is not a problem: TARFS can
125 * still operate without a mutex, but with some limitations.
126 *
127 * In particular, mount() and unmount() will no longer be protected by
128 * the mutex, and this must be taken into account by the platform
129 * implementation.
130 *
131 *
132 * void tarfs_os_init(); --> create a recursive mutex (or no-op)
133 * void tarfs_os_acquire_mutex(); --> acquire the mutex (or no-op)
134 * void tarfs_os_release_mutex(); --> release the mutex (or no-op)
135 *
136 * These functions take no arguments and return no values. The mutex itself
137 * and its implementation must be completely hidden from the filesystem
138 * through this interface.
139 */
140static SemaphoreHandle_t s_lock = NULL;
141
142
146void tarfs_os_init() {
147
148 if (s_lock == NULL)
149 s_lock = xSemaphoreCreateRecursiveMutex();
150}
151
155void tarfs_os_acquire_mutex() {
156
157 if (s_lock != NULL) do {
158 /* absolutely nothing */
159 } while (pdFALSE == xSemaphoreTake(s_lock, portMAX_DELAY));
160}
161
165void tarfs_os_release_mutex() {
166
167 if (s_lock != NULL)
168 xSemaphoreGive(s_lock);
169}

179/*
180 * Default allocator.
181 *
182 * Allocations of 1 KiB or larger are attempted from SPIRAM first
183 * when external memory support is enabled. If SPIRAM is unavailable
184 * or the allocation fails, the default heap is used as a fallback.
185 */
186void *tarfs_os_malloc(size_t size) {
187
188 void *ptr = NULL;
189
190#if CONFIG_TARFS_EXTMEM
191 if (size > 1023) /* TODO: no magic numbers! */
192 ptr = heap_caps_malloc(size, MALLOC_CAP_SPIRAM);
193
194 if (ptr == NULL)
195 /* fallback to the default allocator */
196#endif
197 ptr = heap_caps_malloc(size, MALLOC_CAP_DEFAULT);
198
199 if (ptr == NULL)
200 errno = ENOMEM;
201
202 return ptr;
203}

221void const *tarfs_os_map_tarfile(const char *label, void **os_handle_out, size_t *size_out) {
222
223 esp_partition_iterator_t i;
224 esp_partition_mmap_handle_t handle;
225 const esp_partition_t *part;
226
227 void const *map;
228
229 if (label == NULL || os_handle_out == NULL) {
230 log("ESP32: invalid arguments\r\n");
231 errno = EFAULT;
232 return NULL;
233 }
234
235 /* Fetch pointer to the FLASH partition descriptor by its label */
236 i = esp_partition_find(ESP_PARTITION_TYPE_DATA,
237 ESP_PARTITION_SUBTYPE_ANY,
238 label);
239 if (i == NULL) {
240 log("ESP32: partition not found '%s'\r\n", label);
241 errno = ENOENT;
242 return NULL;
243 }
244
245 part = esp_partition_get(i);
246 esp_partition_iterator_release(i);
247
248 /* non-NULL iterator can not yield NULL partition pointer, but just in case */
249 if (part == NULL) {
250 log("esp_partition_get() returned NULL, this must not happen!\r\n");
251 errno = EIO;
252 return NULL;
253 }
254
255 if (ESP_OK == esp_partition_mmap(part,0,part->size,ESP_PARTITION_MMAP_DATA,&map,&handle)) {
256 *os_handle_out = (void *)handle;
257 if (size_out)
258 *size_out = part->size;
259
260 log("ESP32: partition '%s' -> vaddr=%p, size=%u\r\n", label, map, (unsigned int)part->size);
261
262 return map;
263 }
264
265 log("esp_partition_mmap() failed\r\n");
266 errno = ENOMEM;
267 return NULL;
268}

284/* If the platform provides a VFS (Virtual File System), two additional
285 * functions should be implemented:
286 *
287 * bool tarfs_os_register_fs(const char *prefix, void *context)
288 * bool tarfs_os_unregister_fs(const char *prefix)
289 *
290 * TARFS calls tarfs_os_register_fs() when mounting a filesystem.
291 * This function must inform the platform that the path specified by
292 * prefix is now handled by TARFS.
293 *
294 * The context argument is an opaque value that the VFS will pass as the
295 * first argument to tarf_open(), tarf_close(), tarf_read(), tarf_write(), etc.
296 *
297 * The implementation must make no assumptions about the type or meaning
298 * of context. It may be a pointer, an integer value, or even NULL.
299 */
300
305bool tarfs_os_unregister_fs(const char *prefix) {
306
307 return ESP_OK == esp_vfs_unregister_fs(prefix);
308}

323 ESP_VFS_FLAG_STATIC |

325 context);
